4.2 最小示例：Echo 回声服务器 ​

服务端 examples/node/echo-server.js：

javascript
const { WebSocketServer } = require('ws');

// 创建 WebSocket 服务端，监听 8080 端口
// ws 库内部会自动完成 HTTP Upgrade 握手，无需手动处理
const wss = new WebSocketServer({ port: 8080 });

wss.on('connection', (ws, req) => {
  console.log(`[服务端] 新连接建立，来自 ${req.socket.remoteAddress}，当前在线 ${wss.clients.size} 人`);

  // 连接建立后主动推送 —— 体现 WebSocket 全双工特性
  ws.send('欢迎！你已连接到 Echo 服务器，发送任意内容将被原样返回。');

  // 收到消息原样回显
  ws.on('message', (message) => {
    console.log(`[服务端] 收到消息: ${message}`);
    ws.send(`[echo] ${message}`);
  });

  ws.on('close', (code, reason) => {
    console.log(`[服务端] 连接关闭，code=${code}，reason=${reason || '(空)'}`);
  });

  // 必须监听 error，否则未捕获的错误会让进程崩溃
  ws.on('error', (err) => console.error(`[服务端] 连接出错: ${err.message}`));
});

console.log('[服务端] Echo 服务器已启动，监听 ws://localhost:8080');

Node.js 客户端 examples/node/client.js：

javascript
const WebSocket = require('ws');
const ws = new WebSocket('ws://localhost:8080');

ws.on('open', () => {
  console.log('[客户端] 连接已建立');
  ['你好，WebSocket！', '这是第二条消息', 'bye'].forEach((msg, i) => {
    setTimeout(() => { console.log(`[客户端] 发送: ${msg}`); ws.send(msg); }, i * 500);
  });
});

ws.on('message', (data) => console.log(`[客户端] 收到: ${data}`));
ws.on('close', (code) => { console.log(`[客户端] 连接已关闭，code=${code}`); process.exit(0); });
ws.on('error', (err) => { console.error(`[客户端] 连接出错: ${err.message}`); process.exit(1); });

setTimeout(() => ws.close(1000, '演示结束'), 2500); // 2.5s 后正常关闭

4.4 浏览器客户端（原生 API） ​

浏览器的 WebSocket API 与上面的 Node 客户端几乎一一对应。完整可运行的聊天室页面见 examples/browser/index.html，核心代码：

javascript
// 浏览器原生 API，无需任何库
const ws = new WebSocket('ws://localhost:8080');

ws.onopen = () => { /* 握手成功，更新 UI 状态灯 */ };

// 浏览器端 event.data 直接是字符串（文本帧）或 Blob/ArrayBuffer（二进制帧）
ws.onmessage = (event) => {
  const msg = JSON.parse(event.data);
  appendLog(`[${new Date(msg.time).toLocaleTimeString()}] ${msg.nickname}: ${msg.text}`);
};

ws.onclose = (event) => { /* event.code 是关闭状态码 */ };
ws.onerror = () => { /* 连接失败（如服务端没启动） */ };

// 发送前必须检查 readyState，否则向已断开的连接发送会抛异常
if (ws.readyState === WebSocket.OPEN) {
  ws.send(JSON.stringify({ type: 'chat', text: '你好' }));
}

值常量含义

4.5 聊天室服务端：广播 + 协议级心跳 ​

完整代码见 examples/node/chat-server.js，两个核心机制：

javascript
// ① 广播：遍历所有连接，只向 OPEN 状态的发送
function broadcast(payload) {
  const data = JSON.stringify(payload);
  for (const client of wss.clients) {
    if (client.readyState === WebSocket.OPEN) client.send(data);
  }
}

// ② 协议级心跳：每 10s 发 ping，收不到 pong 的连接强制回收
const heartbeatTimer = setInterval(() => {
  for (const ws of wss.clients) {
    if (ws.isAlive === false) { ws.terminate(); continue; }
    ws.isAlive = false;                 // 先标记为死
    ws.ping();                          // 等 pong 来续命
  }
}, 10_000);

wss.on('connection', (ws) => {
  ws.isAlive = true;
  ws.on('pong', () => { ws.isAlive = true; });  // 收到 pong 即续命
  // ...消息处理
});

为什么需要心跳？ TCP 连接在对方断网、断电、进程崩溃时不会收到任何通知（半开连接）。 没有心跳，服务端会以为客户端还在，白白占用内存和文件描述符。这是生产环境必做项，第 9 章还会深入。
